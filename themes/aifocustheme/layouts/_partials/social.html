{{/* Open Graph and Twitter/X card tags. Replaces Hugo's _internal/opengraph.html and _internal/twitter_cards.html, which on this site emitted no og:title or twitter:title and pointed every page at the same /images/card.jpeg. Essays and tutorials point og:image at /og/.png, a 1200x630 card made by functions/og/[[path]].js from the page's og-card.html output. Every other page keeps the site card. Called with (dict "page" . "desc" $desc) from head.html. */}} {{- $page := .page }} {{- $info := partial "og-card-info.html" $page }} {{- $title := cond $page.IsHome site.Title $info.title }} {{- $desc := .desc | plainify | strings.TrimSpace | truncate 200 }} {{- if $page.IsPage }} {{- with $page.PublishDate }} {{- end }} {{- with $page.Lastmod }} {{- end }} {{- else }} {{- end }} {{- $authorKey := index ($page.Params.authors | default (slice site.Params.defaultAuthor)) 0 }} {{- with index site.Params.authors $authorKey }}{{ with .twitter }} {{- $handle := . | replaceRE `^https?://(www\.)?(x|twitter)\.com/` "" | strings.TrimSuffix "/" }} {{- if findRE `^[A-Za-z0-9_]{1,15}$` $handle }} {{- end }} {{- end }}{{ end }}
